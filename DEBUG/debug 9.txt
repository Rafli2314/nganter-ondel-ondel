# --- in-memory state ---
var player_position: Vector2 = Vector2.ZERO
var ondell_follow_mode: bool = true
var progress_flags: Dictionary = {}   # generic flags: {"door_1_open": true, ...}
var scene_name_before: String = ""

# --- persistence file (optional) ---
const SAVE_PATH := "user://global_state.json"

func _ready():
	# Try to load on startup (optional)
	load_to_memory()
	
	# API: set state before scene change
func set_player_state(pos: Vector2, follow_mode: bool) -> void:
	player_position = pos
	ondell_follow_mode = follow_mode
	
func set_flag(key: String, value) -> void:
	progress_flags[key] = value
	
func get_flag(key: String, default := false):
	if progress_flags.has(key):
		return progress_flags[key]
	return default
	
# Save to disk (optional but useful)
func save_to_disk() -> void:
	var d = {
		"player_position": { "x": player_position.x, "y": player_position.y },
		"ondell_follow_mode": ondell_follow_mode,
		"progress_flags": progress_flags
	}
	var f = FileAccess.open(SAVE_PATH, FileAccess.WRITE)
	if f:
		f.store_string(JSON.stringify(d))
		f.close()

# Load from disk to memory
func load_to_memory() -> void:
	if not FileAccess.file_exists(SAVE_PATH):
		return
	var f = FileAccess.open(SAVE_PATH, FileAccess.READ)
	if not f:
		return
	var txt = f.get_as_text()
	f.close()
	var parsed = JSON.parse_string(txt)
	if typeof(parsed) == TYPE_DICTIONARY:
		var d = parsed
		if d.has("player_position"):
			var p = d["player_position"]
			player_position = Vector2(p.get("x", 0), p.get("y", 0))
		ondell_follow_mode = d.get("ondell_follow_mode", true)
		progress_flags = d.get("progress_flags", {})
		
# Clear saved file
func clear_save() -> void:
	if FileAccess.file_exists(SAVE_PATH):
		DirAccess.remove_absolute(SAVE_PATH)
	player_position = Vector2.ZERO
	ondell_follow_mode = true
	progress_flags.clear()

const SPEED = 120
const JUMP_FORCE = -320.0
var gravity = ProjectSettings.get_setting("physics/2d/default_gravity")
var follow_mode := true

func _ready():
	add_to_group("player")

func _physics_process(delta):
	# Gravity
	if not is_on_floor():
		velocity.y += gravity * delta

	# Horizontal input
	var input_dir = Input.get_axis("ui_left", "ui_right")
	velocity.x = input_dir * SPEED
	if Input.is_action_pressed("ui_right"):
		$AnimatedSprite2D.flip_h = false

	elif Input.is_action_pressed("ui_left"):
		$AnimatedSprite2D.flip_h = true
	# Jump
	if is_on_floor() and Input.is_action_just_pressed("ui_up"):
		velocity.y = JUMP_FORCE
		
	# === ANIMATION STATE MACHINE ===
	if not is_on_floor():
		$AnimatedSprite2D.play("aliya-jump")
	elif velocity.x != 0:
		$AnimatedSprite2D.play("aliya-walk")
	else:
		$AnimatedSprite2D.play("aliya-idle")
		
	# Toggle follow
	if Input.is_action_just_pressed("toggle_follow"):
		var ond = get_node("../ondel-ondel")
		ond.follow_mode = !ond.follow_mode
	move_and_slide()
	
func bounce():
	velocity.y = -200

extends CharacterBody2D

@export var follow_target: NodePath
var speed = 60
var gravity = ProjectSettings.get_setting("physics/2d/default_gravity")
var follow_mode := true

func _physics_process(delta):
	var target = get_node(follow_target)
	
	# Gravity
	if not is_on_floor():
		velocity.y += gravity * delta
	
	# If stay command active → DO NOTHING
	if not follow_mode:
		velocity.x = 0
		$AnimatedSprite2D.play("ondel-ondel-idle")
		move_and_slide()
		return

	# Follow logic
	var dx = target.global_position.x - global_position.x

	if abs(dx) > 16:
		velocity.x = speed * sign(dx)
	else:
		velocity.x = 0

	# Auto-hop if stuck on wall
	if is_on_floor() and is_on_wall():
		velocity.y = -160

	$AnimatedSprite2D.play("ondel-ondel-idle")
	
	move_and_slide()

extends HBoxContainer

@export var line_status: bool = false
var word_input: String = ""

signal word_to_container(word: String)

func _input(event):
	if not line_status:
		return
	
	# ENTER to submit
	if event is InputEventKey and event.is_pressed() and not event.is_echo() and event.keycode == KEY_ENTER:
		emit_signal("word_to_container", word_input)
		return

	# BACKSPACE
	if event is InputEventKey and event.is_pressed() and not event.is_echo() and event.keycode == KEY_BACKSPACE:
		if word_input.length() > 0:
			word_input = word_input.erase(word_input.length() - 1, 1)
			update_label()
		return

	# LETTER INPUT
	if event is InputEventKey and event.is_pressed() and not event.is_echo():
		var ch: String = event.as_text()
		if ch.length() == 1 and word_input.length() < get_child_count():
			word_input += ch
			update_label()
			get_child(word_input.length() - 1).pop_anim()
			print(word_input)

func update_label():
	for i in range(get_child_count()):
		get_child(i).text = ""
	for i in range(word_input.length()):
		get_child(i).text = word_input[i]

var file
var file_dir = "res://scripts/wordle/word list.json"
var data
var today_word
var line_active = 0

func _ready():
	load_data(file_dir)
	today_word = random_word()
	
	get_child(line_active).line_status = true
	connect_to_line()
	print(today_word)

func load_data(dir):
	file = FileAccess.open(dir, FileAccess.READ)
	if file == null:
		push_error("Gagal membuka file")
		return
		
	var text = file.get_as_text()
	
	
	var parsed = JSON.parse_string(text)
	
	if parsed == null:
		push_error("Gagal parse JSON")
		return
	
	data = parsed
	
func random_word():
	var random_id = randi() % data.size()
	var new_word = data[random_id]["word"]
	return new_word.to_upper()

func connect_to_line():
	for i in range(get_child_count()):
		get_child(i).connect("word_to_container", Callable(self, "check_word"))

func check_word(word):
	print("checkword", word)
	if word == today_word:
		check_letter(word)
		emit_signal("wordle_win")
		print("win")
		return
	else:
		for i in range(data.size()):
			if word == data[i]["word"].to_upper():
				print ("valid")
				check_letter(word)
				
func check_letter(word):
	var temp_word:String = today_word
	for i in range(today_word.length()):
		if word[i] == today_word[i]:
			get_child(line_active).get_child(i).change_status(2)
		elif word[i] in temp_word:
			get_child(line_active).get_child(i).change_status(3)
		else:
			get_child(line_active).get_child(i).change_status(1)
		
		temp_word = temp_word.replace(word[i], "")
		
	change_line()
		
func change_line():
	if line_active < get_child_count() - 1:
		get_child(line_active).line_status = false
		line_active += 1
		get_child(line_active).line_status = true
	else:
		print("game over")
		emit_signal("wordle_fail")

func get_meaning(word:String) -> String:
	word = word.to_upper()
	for entry in data:
		if entry["word"] == word:
			return entry["arti"]
	return "Tidak ada arti."
