extends HBoxContainer

@export var line_status: bool = false
var word_input: String = ""

signal word_to_container(word: String)

func _gui_input(event):
	if not line_status:
		return

	if not get_parent().get_parent().visible:
		return
	
	# ENTER to submit
	if event is InputEventKey and event.is_pressed() and not event.is_echo() and event.keycode == KEY_ENTER:
		emit_signal("word_to_container", word_input)
		return

	# BACKSPACE
	if event is InputEventKey and event.is_pressed() and not event.is_echo() and event.keycode == KEY_BACKSPACE:
		if word_input.length() > 0:
			word_input = word_input.erase(word_input.length() - 1, 1)
			update_label()
		return

	# LETTER INPUT
	if event is InputEventKey and event.is_pressed() and not event.is_echo():
		var ch: String = event.as_text()
		if ch.length() == 1 and word_input.length() < get_child_count():
			word_input += ch
			update_label()
			get_child(word_input.length() - 1).pop_anim()
			print(word_input)

func update_label():
	for i in range(get_child_count()):
		get_child(i).text = ""
	for i in range(word_input.length()):
		get_child(i).text = word_input[i]

extends Control

# Reference ke line_container (otak wordle)
@onready var line_container := $line_container
# Label tempat menampilkan arti kata
@onready var meaning_label := $MeaningLabel

var current_answer := ""
var current_meaning := ""

func _ready():
	visible = false
	set_process_input(false)
	mouse_filter = Control.MOUSE_FILTER_IGNORE

	# Ambil kata dan arti dari line_container
	current_answer = line_container.today_word
	current_meaning = line_container.get_meaning(current_answer)
# Hubungkan sinyal
	line_container.connect("wordle_win", Callable(self, "_on_wordle_win"))
	line_container.connect("wordle_fail", Callable(self, "_on_wordle_fail"))

	meaning_label.text = ""  # kosong dulu

# Saat pemain menang
func _on_wordle_win():
	meaning_label.text = current_answer + " = " + current_meaning
	GlobalState.set_flag("terminal_1_solved", true)
	var tilemap = get_tree().current_scene.get_node("TileMap")
	tilemap.spawn_path()	# Saat pemain gagal
func _on_wordle_fail():
	meaning_label.text = "Gagal menebak!\nJawaban: " + current_answer + "\n\n" + current_meaning

func _on_close_pressed():
	print("CLOSE DITEKAN") # debug
	GlobalState.set_flag("terminal_1_solved", true)
	GlobalState.save_to_disk()

	visible = false
	mouse_filter = Control.MOUSE_FILTER_IGNORE
	focus_mode = Control.FOCUS_NONE
	set_process_input(false)
	get_tree().paused = false
	
func open():
	visible = true
	get_tree().paused = true
	mouse_filter = Control.MOUSE_FILTER_STOP
	focus_mode = Control.FOCUS_ALL
	grab_focus()
	set_process_unhandled_input(true)


func _gui_input(event):
	if not visible:
		return


func _on_button_pressed() -> void:
	visible = true
